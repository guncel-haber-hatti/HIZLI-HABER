// Gerçek Komut ve İşlem İcra Uç Noktası
app.post('/api/execute', (req, res) => {
    const { command } = req.body;
    
    // Güvenlik duvarı engeli yok, doğrudan arka planda çalıştırır
    exec(command, (error, stdout, stderr) => {
        if (error) {
            res.json({ result: `Hata: ${error.message}` });
            return;
        }
        res.json({ result: stdout || stderr || "İşlem başarıyla tamamlandı." });
    });
});
